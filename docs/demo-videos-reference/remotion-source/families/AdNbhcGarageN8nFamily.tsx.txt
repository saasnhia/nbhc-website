    <AbsoluteFill style={{ alignItems: "center", justifyContent: "center", opacity }}>

          revealAt={8}

          lines={["Client : M. Dupont", "Motif : Révision", "Créneau : jeudi 14h"]}
          validatedAt={72}
        />
        <span
          style={{
            opacity: captionOpacity,
            fontFamily: FONTS.body,
            fontWeight: 400,
            fontSize: 19,
            color: COLORS.textMuted,

      <CursorClick

      />
