//   S4 (450-570)  Usage par le pro — ValidationCard + CursorClick
//   S5 (570-630)  Installation en 3 étapes

import React from "react";
import { AbsoluteFill, interpolate, Sequence, useCurrentFrame } from "remotion";
import { COLORS, FONTS } from "../tokens";
import { PhoneCall } from "../components/PhoneCall";
import { VoiceBubble } from "../components/VoiceBubble";
import { ValidationCard } from "../components/ValidationCard";
import { N8nCanvas } from "../components/N8nCanvas";
import { SoftwareBadges } from "../components/SoftwareBadges";
import { InstallSteps } from "../components/InstallSteps";
import { CursorClick } from "../../components/CursorClick";
import { MegaCapsFinale } from "../../marketing/helpers/MegaCapsFinale";

import { PhoneIcon, WaveformIcon, SparkIcon, CalendarIcon, CalendarCheckIcon, FilePlusIcon } from "../icons";

const SceneHook: React.FC = () => <HookTitle text="Pendant une coupe, qui décroche le téléphone ?" />;

const SceneDeclencheur: React.FC = () => {
  const frame = useCurrentFrame();
  const opacity = edgeFade(frame, 90);
  return (
    <AbsoluteFill style={{ alignItems: "center", justifyContent: "center", opacity }}>
      <PhoneCall localFrame={frame} ringUntil={45} callerName="Client" bascuLabel="Votre ligne actuelle, renvoi activé." />
    </AbsoluteFill>
  );
};

  { label: "IA Mistral : comprend la demande", icon: <SparkIcon /> },
  { label: "Vérifie le planning", icon: <CalendarIcon /> },

  { label: "Crée la fiche RDV", icon: <FilePlusIcon /> },
];

  const frame = useCurrentFrame();
  const opacity = edgeFade(frame, 210);
  return (
    <AbsoluteFill style={{ alignItems: "center", justifyContent: "center", opacity }}>
      <div style={{ display: "flex", flexDirection: "column", alignItems: "center", gap: 36 }}>
        <N8nCanvas localFrame={frame} nodes={WORKFLOW_NODES} activateAt={WORKFLOW_ACTIVATE_AT} />
        <VoiceBubble

            { speaker: "ia", text: "Samedi 11h ?", revealAt: 95 },
            { speaker: "client", text: "Parfait.", revealAt: 135 },

        />
      </div>
    </AbsoluteFill>
  );
};

const SceneLogiciels: React.FC = () => {
  const frame = useCurrentFrame();
  const opacity = edgeFade(frame, 90);
  return (
    <AbsoluteFill style={{ alignItems: "center", justifyContent: "center", opacity }}>

  );

const SceneUsage: React.FC = () => {
  const frame = useCurrentFrame();
  const opacity = edgeFade(frame, 120);
  const captionOpacity = interpolate(frame, [85, 97], [0, 1], { extrapolateLeft: "clamp", extrapolateRight: "clamp" });
  return (
    <AbsoluteFill style={{ alignItems: "center", justifyContent: "center", opacity }}>
      <div style={{ display: "flex", flexDirection: "column", alignItems: "center", gap: 24 }}>
        <ValidationCard
          localFrame={frame}
          revealAt={8}
          title="Nouveau RDV"
          lines={["Cliente : Mme Martin", "Motif : Coupe", "Créneau : samedi 11h"]}
          validatedAt={72}
        />
        <span
          style={{
            opacity: captionOpacity,
            fontFamily: FONTS.body,
            fontWeight: 400,
            fontSize: 19,
            color: COLORS.textMuted,
          }}
        >
          Vous validez d'un clic, quand vous voulez.
        </span>
      </div>
      <CursorClick
        target_x={57}
        target_y={58}
        origin="right"
        durationInFrames={120}
        entry_ms={1400}
        click_at_ms={2400}
        cursor_style="arrow"
        accent_color={COLORS.accent}
        size_multiplier={1}
        ripple_on_click={true}
        exit_strategy="fade"
      />
    </AbsoluteFill>
  );
};
