const PATIENT_ROWS = [
  { label: "M. D. · 62 ans", eligible: false },
  { label: "Mme R. · 71 ans", eligible: true },
  { label: "J. P. · 45 ans", eligible: false },
  { label: "A. K. · 68 ans", eligible: true },

];

const SceneListe: React.FC = () => {
  const frame = useCurrentFrame();
  const opacity = edgeFade(frame, 150);
  const highlightOn = frame >= 90;
  const captionOpacity = interpolate(frame, [100, 115], [0, 1], { extrapolateLeft: "clamp", extrapolateRight: "clamp" });
  const criteriaOpacity = interpolate(frame, [118, 133], [0, 1], { extrapolateLeft: "clamp", extrapolateRight: "clamp" });

  return (
    <AbsoluteFill style={{ alignItems: "center", justifyContent: "center", opacity }}>
      <div style={{ display: "flex", flexDirection: "column", alignItems: "center", gap: 20 }}>
        <div
          style={{
            width: 560,
            borderRadius: 20,
            background: COLORS.glassDark,
            border: `1px solid ${COLORS.glassDarkBorder}`,
            boxShadow: "0 16px 40px rgba(0,0,0,0.45)",
            padding: 38,
          }}
        >
          <div style={{ fontFamily: FONTS.display, fontWeight: 500, fontSize: 22, letterSpacing: "0.01em", color: COLORS.text, marginBottom: 18 }}>
            Patientèle
          </div>
          {PATIENT_ROWS.map((row, i) => {
            const rowDelay = i * 6;
            const local = frame - rowDelay;
            if (local < 0) return null;
            const rowOpacity = interpolate(local, [0, 8], [0, 1], { extrapolateRight: "clamp" });
            const isHighlighted = highlightOn && row.eligible;
            const highlightT = interpolate(frame, [90, 100], [0, 1], { extrapolateLeft: "clamp", extrapolateRight: "clamp" });
            return (
              <div
                key={row.label}
                style={{
                  opacity: rowOpacity,
                  padding: "11px 16px",
                  margin: "3px 0",
                  borderRadius: 8,
                  border: `1px solid ${isHighlighted ? COLORS.accent : "transparent"}`,
                  background: isHighlighted ? `rgba(10,132,255,${0.12 * highlightT})` : "transparent",
                  fontFamily: FONTS.body,
                  fontWeight: 400,
                  fontSize: 20,
                  color: isHighlighted ? COLORS.text : COLORS.textMuted,
                }}
              >
                {row.label}
              </div>
            );
          })}
        </div>
        <span
          style={{
            opacity: captionOpacity,
            fontFamily: FONTS.display,
            fontWeight: 500,
            fontSize: 21,
            letterSpacing: "0.01em",
            color: COLORS.accent,
          }}
        >
          Éligible au bilan partagé de médication
        </span>
        <span
          style={{
            opacity: criteriaOpacity,
            fontFamily: FONTS.body,
            fontWeight: 400,
            fontSize: 18,
            color: COLORS.textDim,
          }}
        >
          Critères : 65 ans et +, 5 traitements chroniques ou plus
        </span>
      </div>
    </AbsoluteFill>
  );
};
