// Reconstitution stylisée de l'éditeur n8n (canvas + nodes reliés par des
// courbes) à la charte nbhc — pas la vraie UI violette de n8n, mais un
// schéma reconnaissable comme "un workflow de nodes reliés", cohérent avec
// le reste des vidéos (glassmorphism sobre, accent unique cf tokens.ts).
//
// Les nodes s'activent en séquence (accent + halo) et une impulsion se
// dessine le long de chaque courbe entre deux nodes, gauche → droite, pour
// représenter le flux de données qui circule.

import React from "react";
import { interpolate } from "remotion";
import { COLORS, FONTS } from "../tokens";

export type N8nNodeSpec = {
  label: string;
  icon: React.ReactNode;
};

type N8nCanvasProps = {
  localFrame: number;
  nodes: N8nNodeSpec[];
  /** Frame à laquelle chaque node s'active — même longueur que `nodes`, croissant. */
  activateAt: number[];
  /** Largeur du canvas — par défaut calculée pour épouser le contenu (pas de vide latéral). */
  width?: number;
};

const NODE_WIDTH = 244;
const NODE_HEIGHT = 118;
const GAP = 42;
const CANVAS_HEIGHT = 440;
const CANVAS_PADDING_X = 64;
const ZIGZAG_OFFSET = 60;

export const N8nCanvas: React.FC<N8nCanvasProps> = ({ localFrame, nodes, activateAt, width }) => {
  const n = nodes.length;
  const totalWidth = n * NODE_WIDTH + (n - 1) * GAP;
  const canvasWidth = width ?? totalWidth + CANVAS_PADDING_X * 2;
  const startX = (canvasWidth - totalWidth) / 2;
  const centerY = CANVAS_HEIGHT / 2 - NODE_HEIGHT / 2;
  const positions = nodes.map((_, i) => ({
    x: startX + i * (NODE_WIDTH + GAP),
    y: centerY + (i % 2 === 0 ? -ZIGZAG_OFFSET : ZIGZAG_OFFSET),
  }));

  return (
    <div
      style={{
        position: "relative",
        width: canvasWidth,
        height: CANVAS_HEIGHT,
        borderRadius: 24,
        background: "rgba(28,28,30,0.65)",
        border: "1.5px solid rgba(255,255,255,0.14)",
        boxShadow: "0 20px 50px rgba(0,0,0,0.4)",
      }}
    >
      <svg width={canvasWidth} height={CANVAS_HEIGHT} style={{ position: "absolute", inset: 0 }}>
        {positions.slice(0, -1).map((p, i) => {
          const next = positions[i + 1];
          const x1 = p.x + NODE_WIDTH;
          const y1 = p.y + NODE_HEIGHT / 2;
          const x2 = next.x;
          const y2 = next.y + NODE_HEIGHT / 2;
          const midX = (x1 + x2) / 2;
          const path = `M ${x1} ${y1} C ${midX} ${y1}, ${midX} ${y2}, ${x2} ${y2}`;

          const segStart = activateAt[i];
          const segEnd = activateAt[i + 1];
          const progress = interpolate(localFrame, [segStart, segEnd], [0, 1], {
            extrapolateLeft: "clamp",
            extrapolateRight: "clamp",
          });
          const lit = localFrame >= segStart;

          return (
            <g key={i}>
              <path d={path} fill="none" stroke={COLORS.glassDarkBorder} strokeWidth={2} />
              {lit && (
                <path
                  d={path}
                  fill="none"
                  stroke={COLORS.accent}
                  strokeWidth={2.5}
                  pathLength={1}
                  strokeDasharray={1}
                  strokeDashoffset={1 - progress}
                  opacity={0.95}
                />
              )}
            </g>
          );
        })}
      </svg>

      {nodes.map((node, i) => {
        const p = positions[i];
        const local = localFrame - activateAt[i];
        const isActive = local >= 0;
        const scale = interpolate(local, [0, 12], [0.94, 1], {
          extrapolateLeft: "clamp",
          extrapolateRight: "clamp",
        });
        const glow = isActive ? interpolate(local, [0, 16], [0, 1], { extrapolateRight: "clamp" }) : 0;
        return (
          <div
            key={i}
            style={{
              position: "absolute",
              left: p.x,
              top: p.y,
              width: NODE_WIDTH,
              minHeight: NODE_HEIGHT,
              transform: `scale(${scale})`,
              borderRadius: 14,
              background: COLORS.glassDark,
              border: `1.5px solid ${isActive ? COLORS.accent : COLORS.glassDarkBorder}`,
              boxShadow: isActive ? `0 0 ${10 * glow}px ${COLORS.accentGlow}` : "none",
              display: "flex",
              alignItems: "center",
              gap: 16,
              padding: "20px 26px",
            }}
          >
            <div style={{ color: isActive ? COLORS.accent : COLORS.textMuted, opacity: isActive ? 1 : 0.55, flexShrink: 0 }}>
              {node.icon}
            </div>
            <span
              style={{
                fontFamily: FONTS.body,
                fontWeight: 400,
                fontSize: 23,
                lineHeight: 1.3,
                letterSpacing: "0.005em",
                color: isActive ? COLORS.text : COLORS.textMuted,
              }}
            >
              {node.label}
            </span>
          </div>
        );
      })}
    </div>
  );
};
